48 @property # Returns the postal code extracted from the full address for privacy. 

49 def search_location(self) -> str: 

50 """Only the postal code of the address is used for searching (privacy).""" 

51 match = re.search(r"\b\d{5}\b", self.address) # Only the postal code is extracted for privacy. 

52 return match.group() if match else ""

139class ScoredJob(BaseModel): 

140 job: JobPosting 

141 match: JobMatch 

142 # Commute time in minutes per travel mode; modes without a route are missing. 

143 commute_minutes: dict[TravelMode, int] = Field(default_factory=dict) 

144 

145 @property 

146 def fastest_commute(self) -> tuple[TravelMode, int] | None: 

147 if not self.commute_minutes: 

148 return None 

149 return min(self.commute_minutes.items(), key=lambda item: item[1])

152class CoverLetter(BaseModel): 

153 """A cover letter draft. The sender's address block is added by code, not by the model.""" 

154 

155 subject: str = Field(description="Subject line, e.g. 'Bewerbung als Werkstudent Softwareentwicklung'.") 

156 salutation: str = Field(description="Use the contact person's name if the posting names one.") 

157 paragraphs: list[str] = Field(description="3-5 paragraphs of body text.") 

158 closing: str = Field(description="Only the closing formula, e.g. 'Mit freundlichen Grüßen' - no name.") 

159 

160 # Convert the cover letter to plain text with sections separated by double newlines. 

161 def as_text(self) -> str: 

162 return "\n\n".join([self.subject, self.salutation, *self.paragraphs, self.closing])
