47 # Drop ids the model may have invented - never trust LLM output blindly.

50 # for a new assessment. Their status (e.g. "applied") is kept. 

51 known = {stored.scored.job.id: stored.scored for stored in load_jobs([job.id for job in shortlist])} 

52 to_assess = [job for job in shortlist if job.id not in known] 

53 print(

55 f"({len(known)} already assessed earlier). Assessing ...\n"

57

59 

60 save_jobs(new_jobs)

62 

63 for number, scored in enumerate(scored_jobs, start=1):

65 type_warning = "" if match.fits_employment_type else " [wrong employment type]"

98async def run_cover_letter(job_id: str) -> None: 

99 """Separate workflow: write a letter for a posting found in an earlier search run.""" 

100 stored = load_jobs([job_id]) 

101 if not stored: 

102 sys.exit(f"No posting with ID {job_id!r} in the database. Copy the ID from the Excel file.") 

103 preferences = load_preferences() 

104 if preferences is None: # the sender address comes from the setup 

105 sys.exit("No preferences yet - run `uv run application-agent setup` first.") 

106 profile = await get_profile() 

107 scored = stored[0].scored 

108 print(f"Cover letter for: {scored.job.title} | {scored.job.company}\n") 

109 

110 print("Anything the letter should mention? (e.g. 'available from 01.12.', Enter for none)") 

111 notes = input("> ").strip() 

112 

113 letter, reviews = await write_cover_letter(profile, preferences, scored, notes) 

114 path = save_cover_letter(letter, profile, preferences, scored) 

115 

116 print(f"\n{letter.as_text()}\n") 

117 last_review = reviews[-1] 

118 if not last_review.approved: 

119 print("The reviewer did not approve the final version. Please check:") 

120 for item in last_review.unsupported_claims + last_review.feedback: 

121 print(f" - {item}") 

122 print(f"\nSaved to {path}") 

123 print(f"Apply here: {scored.job.application_url or scored.job.url}")

132def _argument(index: int, prompt: str) -> str: 

133 """Command line argument at ``index``, or ask for it if it is missing.""" 

134 value = sys.argv[index] if len(sys.argv) > index else input(prompt) 

135 # Copied cells sometimes bring spaces or quotes along. 

136 return value.strip().strip("'\"")

145def job_status_main() -> None:

147 statuses = ", ".join(JobStatus) 

148 job_id = _argument(1, "Job ID (from the Excel file): ") 

149 value = _argument(2, f"New status ({statuses}): ").lower()
