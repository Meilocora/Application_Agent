1"""Function tools the agents can call. 

2 

3The docstring and type hints of each tool are sent to the model: they ARE the tool's 

4documentation, from which the model decides when and how to call it. 

5""" 

6 

7import httpx 

8from agents import RunContextWrapper, function_tool 

9 

10from application_agent.arbeitsagentur import search_jobs 

11from application_agent.context import JobSearchContext 

12 

13 

14@function_tool 

15async def search_arbeitsagentur( 

16 wrapper: RunContextWrapper[JobSearchContext], 

17 what: str, 

18 where: str | None = None, 

19) -> str: 

20 """Search the job board of the German Bundesagentur fuer Arbeit (largest job board in Germany). 

21 

22 Employment type and search radius are applied automatically from the user's preferences. 

23 

24 Args: 

25 what: Job title or keyword in German or English, e.g. "Python Entwickler". Keep it short (1-3 words). 

26 where: German city name (e.g. "München", not "Munich") or postal code. Use null for a nationwide search. 

27 """ 

28 # Hard constraints come from the user's preferences, not from the model. 

29 preferences = wrapper.context.preferences 

30 try: 

31 jobs = await search_jobs(what, where, preferences.radius_km, preferences.employment_type) 

32 except httpx.HTTPError as exc: 

33 # Returned to the model as text, so it can react (e.g. retry with other keywords). 

34 return f"Error calling the Arbeitsagentur API: {exc}" 

35 

36 new_jobs = wrapper.context.add_postings(jobs) 

37 if not new_jobs: 

38 return "No new results." 

39 # "id=..." instead of "[...]": with brackets the model sometimes returned them as part of the id. 

40 return "\n".join(f"id={job.id} | {job.title} | {job.company} | {job.location}" for job in new_jobs)
